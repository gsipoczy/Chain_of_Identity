* | Instance Public Method ZCL_IM_C_COI_PROCESS_ORDER->IF_EX_WORKORDER_UPDATE~AT_SAVE
* +-------------------------------------------------------------------------------------------------+
* | [--->] IS_HEADER_DIALOG               TYPE        COBAI_S_HEADER_DIALOG
* | [EXC!] ERROR_WITH_MESSAGE
* +--------------------------------------------------------------------------------------</SIGNATURE>
  METHOD if_ex_workorder_update~at_save.

    DATA: ls_caufvdb TYPE caufvdb,
          lv_needed  TYPE xfeld,
          lv_msgty   TYPE syst_msgty.

    CHECK is_header_dialog-zzcoi = space.

    MOVE-CORRESPONDING is_header_dialog TO ls_caufvdb.
    CALL METHOD zcl_mm_coi_utility=>check_coi_needed_po
      EXPORTING
        is_caufvdb = ls_caufvdb
      IMPORTING
        ev_needed  = lv_needed.
    CHECK lv_needed = abap_true.

    " If not released: warning, if released, error
    IF is_header_dialog-ftrmi IS INITIAL.
      lv_msgty = 'W'.
    ELSE.
      lv_msgty = 'E'.
    ENDIF.

    CALL FUNCTION 'C14Z_MESSAGES_SHOW_AS_POPUP'
      EXPORTING
        i_msgid = 'ZMM_COI_MSG'
        i_msgty = lv_msgty
        i_msgno = '015'.


  ENDMETHOD.

* | [--->] IT_PIINSTRUCTIONVALUE          TYPE        COBAI_T_PIINSTRUCTIONVALUE(optional)
* +--------------------------------------------------------------------------------------</SIGNATURE>
  METHOD if_ex_workorder_update~in_update.

    DATA: ls_index TYPE zpp_coi_index.

    FIELD-SYMBOLS: <lfs_header> TYPE cobai_s_header,
                   <lfs_item>   TYPE cobai_s_item.

    IF it_header IS INITIAL.
      EXIT.
    ENDIF.
    LOOP AT it_header ASSIGNING <lfs_header>.

      " If COI is filled, update batches
      IF <lfs_header>-zzcoi <> space AND <lfs_header>-aufnr <> space.

        " Create index record
        CLEAR ls_index.
        ls_index-mandt = sy-mandt.
        ls_index-coi = <lfs_header>-zzcoi.
        ls_index-aufnr = <lfs_header>-aufnr.
        CALL METHOD zcl_pp_coi_index=>save_index_wrapper
          EXPORTING
            is_index                = ls_index
            iv_delold_process_order = abap_true.
*            iv_delold_sales_order   = ' '
*            iv_delold_batch         = ' '
*            iv_commit               = ' '
*            iv_background           = ' '


        " Update batches
        LOOP AT it_item ASSIGNING <lfs_item>
          WHERE aufnr = <lfs_header>-aufnr.

          CALL METHOD zcl_mm_coi_utility=>update_batch_coi_din
            EXPORTING
              iv_matnr  = <lfs_item>-matnr
              iv_charg  = <lfs_item>-charg
              iv_coi    = <lfs_header>-zzcoi
              iv_din    = space
              iv_commit = abap_false.

        ENDLOOP.

      ENDIF.

    ENDLOOP.

  ENDMETHOD.
