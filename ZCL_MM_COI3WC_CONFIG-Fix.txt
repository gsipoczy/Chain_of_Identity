class ZCL_MM_COI3WC_CONFIG definition

* | Static Public Method ZCL_MM_COI3WC_CONFIG=>GET_CONFIGURATION
* +-------------------------------------------------------------------------------------------------+
* | [--->] IV_MATNR                       TYPE        MATNR
* | [<---] ET_CONFIGURATION               TYPE        ZMM_COI3WC_CONF_SCREEN_TT

    " Get the material info - actually to see which screen is wanted
    CALL METHOD zcl_mm_coi_utility=>get_material_information

        EXPORTING

          it_param     = lt_param
        IMPORTING
          ev_field_obl = ls_field-field_obl.
      ls_field-checksum = abap_false.

      ls_field-value = space.
      APPEND ls_field TO ls_screen-fields.

      ls_field-ord = '05'.
      ls_field-tabname = zcl_mm_coi_utility=>screen_table.
      ls_field-fieldname = zcl_mm_coi_utility=>screenfield_input.
      ls_field-fieldtext = zcl_mm_coi_utility=>zfs_fieldname_collc.

      ls_field-checksum = abap_false.
      ls_field-atnam = zcl_mm_coi_utility=>zfs_atnam_collc_a.
      ls_field-value = space.

    ENDIF. " END of SCREEN 1

        IMPORTING
          ev_field_obl = ls_field-field_obl.
      ls_field-checksum = abap_true.
      ls_field-is_din = abap_true.
      ls_field-atnam = zcl_mm_coi_utility=>zfs_atnam_din_b.
      ls_field-value = space.

      CLEAR ls_field.
      ls_field-ord = '04'.
      ls_field-tabname = zcl_mm_coi_utility=>screen_table.
      ls_field-fieldname = zcl_mm_coi_utility=>screenfield_input.
      ls_field-fieldtext = zcl_mm_coi_utility=>zfs_fieldname_bagid.
      CALL METHOD zcl_mm_coi_utility=>zfs_decide_field_obligatory

        IMPORTING

      ls_field-checksum = abap_false.
      ls_field-atnam = zcl_mm_coi_utility=>zfs_atnam_bagid_b.
      ls_field-value = space.
      APPEND ls_field TO ls_screen-fields.

      CLEAR ls_field.
      ls_field-ord = '05'.
      ls_field-tabname = zcl_mm_coi_utility=>screen_table.
      ls_field-fieldname = zcl_mm_coi_utility=>screenfield_input.
      ls_field-fieldtext = zcl_mm_coi_utility=>zfs_fieldname_collc.

          it_param     = lt_param
        IMPORTING
          ev_field_obl = ls_field-field_obl.
      ls_field-checksum = abap_false.
      ls_field-atnam = zcl_mm_coi_utility=>zfs_atnam_collc_b.
      ls_field-value = space.
      APPEND ls_field TO ls_screen-fields.

      APPEND ls_screen TO lt_conf.
    ENDIF. " END of SCREEN 2

    " Screen 3
    IF lv_screen_c = zcl_mm_coi_utility=>zfs_fixval_atwrt_yes.
      CLEAR ls_screen.
      ls_screen-ord = '03'.
      ls_screen-popup_title = text_title_screen_c.

      " Fields
      CLEAR ls_field.
      ls_field-ord = '01'.
      ls_field-tabname = zcl_mm_coi_utility=>screen_table.
      ls_field-fieldname = zcl_mm_coi_utility=>screenfield_input.
      ls_field-fieldtext = zcl_mm_coi_utility=>zfs_fieldname_patid.

        IMPORTING
          ev_field_obl = ls_field-field_obl.

      ls_field-fieldname = zcl_mm_coi_utility=>screenfield_input.
      ls_field-fieldtext = zcl_mm_coi_utility=>zfs_fieldname_coi.
      CALL METHOD zcl_mm_coi_utility=>zfs_decide_field_obligatory

          it_param     = lt_param
        IMPORTING
          ev_field_obl = ls_field-field_obl.

      ls_field-atnam = zcl_mm_coi_utility=>zfs_atnam_coi_c.

      APPEND ls_field TO ls_screen-fields.

      ls_field-ord = '03'.
      ls_field-tabname = zcl_mm_coi_utility=>screen_table.

        IMPORTING

      ls_field-checksum = abap_true.

      ls_field-fieldtext = zcl_mm_coi_utility=>zfs_fieldname_bagid.

      APPEND ls_field TO ls_screen-fields.

      ls_field-fieldname = zcl_mm_coi_utility=>screenfield_input.
      ls_field-fieldtext = zcl_mm_coi_utility=>zfs_fieldname_collc.

        IMPORTING
          ev_field_obl = ls_field-field_obl.
      ls_field-checksum = abap_false.

    ENDIF. " END of SCREEN 2

    ev_error = abap_false.
    et_configuration = lt_conf.

ENDCLASS.
